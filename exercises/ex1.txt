int main(void) {
	cybsp_init();
	__enable_irq();

	bool led_state = false;
	bool prev_state1 = 1;
	bool prev_state2 = 1;

	while (1) {

		bool curr_state1 = Cy_GPIO_Read(GPIO_PRT8, 3);
		bool curr_state2 = Cy_GPIO_Read(GPIO_PRT8, 7);

		/*
				if(Cy_GPIO_Read(GPIO_PRT8, 3) == 0)
					{
						Cy_GPIO_Clr(GPIO_PRT16, 6);

				}*/

		/* Detect falling edge (button press) */
		if (prev_state1 == 1 && curr_state1 == 0) {

			Cy_SysLib_Delay(20); // debounce

			if (Cy_GPIO_Read(GPIO_PRT8, 3) == 0) {
				led_state = !led_state;
				Cy_GPIO_Write(GPIO_PRT16, 7, led_state);
				Cy_GPIO_Clr(GPIO_PRT16, 6);
			}
		}
		/* Detect falling edge (button press) */
		if (prev_state2 == 1 && curr_state2 == 0) {
			Cy_SysLib_Delay(20); // debounce

			if (Cy_GPIO_Read(GPIO_PRT8, 7) == 0) {
				led_state = !led_state;
				Cy_GPIO_Write(GPIO_PRT16, 6, led_state);
				Cy_GPIO_Clr(GPIO_PRT16, 7);
			}
			// Cy_GPIO_Clr(GPIO_PRT16, 7);
		}

		prev_state1 = curr_state1;
		prev_state2 = curr_state2;
	}
}
