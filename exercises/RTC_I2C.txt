#include "cy_pdl.h"
#include "cybsp.h"
#include "retarget_io_init.h"
#include <stdio.h>
#include "cycfg_peripherals.h" // Includes RTC_HW and RTC_config
#include"cy_rtc.h"


#define DS3231_ADDR (0x68u)
#define RTC_HW SCB0
#define CY_RETARGET_IO_BAUDRATE 115200
#define CYBSP_DEBUG_UART_TX PRT6[5]
// I2C context
cy_stc_scb_i2c_context_t i2c_context;

// Time structure
typedef struct {
    uint8_t seconds;
    uint8_t minutes;
    uint8_t hours;
    uint8_t day;
    uint8_t date;
    uint8_t month;
    uint8_t year;
} rtc_time_t;

// BCD conversion
uint8_t dec_to_bcd(uint8_t val) {
    return ((val / 10) << 4) | (val % 10);
}

uint8_t bcd_to_dec(uint8_t val) {
    return ((val >> 4) * 10) + (val & 0x0F);
}




// Reset I2C bus in case of a stall
void i2c_reset(void) {
    Cy_SCB_I2C_Disable(RTC_HW, &i2c_context);
    Cy_SysLib_Delay(10);
    Cy_SCB_I2C_Enable(RTC_HW);
    printf("I2C Bus Reset Performed\r\n");
}
void i2c_isr(void)
{
    Cy_SCB_I2C_Interrupt(RTC_HW, &i2c_context);
}

// Write time to DS3231
bool ds3231_write_time(rtc_time_t *time) {
    uint8_t data[8];
    data[0] = 0x00; // Start at seconds register
    data[1] = dec_to_bcd(time->seconds) & 0x7F;       // Ensure CH (clock halt) bit = 0
    data[2] = dec_to_bcd(time->minutes);
    data[3] = dec_to_bcd(time->hours) & 0x3F;         // Ensure 24-hour format
    data[4] = dec_to_bcd(time->day);
    data[5] = dec_to_bcd(time->date);
    data[6] = dec_to_bcd(time->month);
    data[7] = dec_to_bcd(time->year);

    cy_stc_scb_i2c_master_xfer_config_t write_config = {
        .slaveAddress = DS3231_ADDR,
        .buffer = data,
        .bufferSize = sizeof(data),
        .xferPending = false // End with STOP
    };

    cy_en_scb_i2c_status_t status = Cy_SCB_I2C_MasterWrite(RTC_HW, &write_config, &i2c_context);
    if (status != CY_SCB_I2C_SUCCESS) return false;

    // Wait for transfer to complete
    while (0UL != (CY_SCB_I2C_MASTER_BUSY & Cy_SCB_I2C_MasterGetStatus(RTC_HW, &i2c_context)));

    return true;
}

// Read time from DS3231
bool ds3231_read_time(rtc_time_t *time) {
    uint8_t reg_addr = 0x00;
    uint8_t data[7] = {0};

    // Write register address
    cy_stc_scb_i2c_master_xfer_config_t reg_config = {
        .slaveAddress = DS3231_ADDR,
        .buffer = &reg_addr,
        .bufferSize = 1,
        .xferPending = true // Repeated start
    };

    cy_en_scb_i2c_status_t status = Cy_SCB_I2C_MasterWrite(RTC_HW, &reg_config, &i2c_context);
    if (status != CY_SCB_I2C_SUCCESS) return false;
    while (0UL != (CY_SCB_I2C_MASTER_BUSY & Cy_SCB_I2C_MasterGetStatus(RTC_HW, &i2c_context)));

    // Read time data
    cy_stc_scb_i2c_master_xfer_config_t read_config = {
        .slaveAddress = DS3231_ADDR,
        .buffer = data,
        .bufferSize = sizeof(data),
        .xferPending = false
    };

    status = Cy_SCB_I2C_MasterRead(RTC_HW, &read_config, &i2c_context);
    if (status != CY_SCB_I2C_SUCCESS) return false;
    while (0UL != (CY_SCB_I2C_MASTER_BUSY & Cy_SCB_I2C_MasterGetStatus(RTC_HW, &i2c_context)));

    time->seconds = bcd_to_dec(data[0] & 0x7F);
    time->minutes = bcd_to_dec(data[1]);
    time->hours   = bcd_to_dec(data[2] & 0x3F);
    time->day     = bcd_to_dec(data[3]);
    time->date    = bcd_to_dec(data[4]);
    time->month   = bcd_to_dec(data[5] & 0x1F);
    time->year    = bcd_to_dec(data[6]);

    return true;
}

int main(void) {
    cy_rslt_t result;

	/* Initialize the device and board peripherals */
	result = cybsp_init();

	/* Board init failed. Stop program execution */
	if (CY_RSLT_SUCCESS != result) {
		handle_app_error();
	}

	/* Enable global interrupts */
	__enable_irq();
	init_retarget_io();
    printf("DS3231 RTC with SCB I2C\r\n");

    Cy_SCB_I2C_Init(RTC_HW, &CYBSP_I2C_CONTROLLER_config, &i2c_context);
    Cy_SCB_I2C_Enable(RTC_HW);

    // Setup ISR
    cy_stc_sysint_t i2c_intr_config = {
        .intrSrc = scb_0_interrupt_IRQn, // Replace with your actual SCB
        .intrPriority = 3
    };
    Cy_SysInt_Init(&i2c_intr_config, i2c_isr);
    NVIC_EnableIRQ(i2c_intr_config.intrSrc);
    // Set time: HH:MM:SS, Day (1=Sun), Date, Month, Year (last two digits)
    rtc_time_t set_time = {0, 21, 19, 4, 15, 5, 25};
    if (ds3231_write_time(&set_time)) {
        printf("Time Set Successfully\r\n");
    } else {
        printf("Failed to Set Time\r\n");
    }

    Cy_SysLib_Delay(500);

    rtc_time_t current_time = {0};

    for (;;) {
        if (ds3231_read_time(&current_time)) {
            printf("Time: %02d:%02d:%02d Date: %02d/%02d/20%02d\r\n",
                   current_time.hours,
                   current_time.minutes,
                   current_time.seconds,
                   current_time.date,
                   current_time.month,
                   current_time.year);
        } else {
            printf("Failed to read DS3231 time!\r\n");
        }

        Cy_SysLib_Delay(1000);
    }
}
