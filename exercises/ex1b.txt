#include "cybsp.h"
#include "cy_time.h"
#include "FreeRTOS.h"
#include "task.h"
#include "cyabs_rtos.h"
#include "cyabs_rtos_impl.h"
#include "cy_pdl.h"
#include "cybsp.h"
#include "cy_syslib.h"

int main(void)
{
    cybsp_init();
    __enable_irq();

   
    bool led_state = false;
    bool prev_state = 1;

    while(1)
    {Cy_GPIO_Write(GPIO_PRT16, 6, 1);
   
            bool curr_state = Cy_GPIO_Read(GPIO_PRT8, 3);

        /* Detect falling edge (button press) */
        if(prev_state == 1 && curr_state == 0)
        {
            Cy_SysLib_Delay(20);  // debounce

            if(Cy_GPIO_Read(GPIO_PRT8, 3) == 0)
            {
                led_state = !led_state;
                Cy_GPIO_Write(GPIO_PRT16, 7, led_state);
            }
       }

        prev_state = curr_state;
    }
}
